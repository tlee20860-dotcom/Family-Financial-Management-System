// ============================================ // fixed-expenses.js — 家庭固定支出（模板 + 每月狀態） // ============================================ import { listenFixedTemplates, addFixedTemplate, updateFixedTemplate, removeFixedTemplate, listenFixedStatus, saveFixedStatus, } from './db.js'; import { formatHKD, escapeHtml } from './utils.js'; import { AppState } from './state.js'; let templates = []; let statuses = {}; let editingId = null; let unsubStatus = null; export function initFixedExpensesPage() { const tbody = document.getElementById('fixed-tbody'); const modal = document.getElementById('fixed-modal'); const modalTitle = document.getElementById('fixed-modal-title'); const form = document.getElementById('fixed-form'); const nameInput = document.getElementById('fixed-name'); const amountInput = document.getElementById('fixed-amount'); const cycleSelect = document.getElementById('fixed-cycle'); const noteInput = document.getElementById('fixed-note'); // 1. 監聽所有固定支出模板（排除成員個人模板） listenFixedTemplates((list) => { templates = list.filter((t) => !t.memberId && t.type !== 'member'); render(); }); // 2. 監聽當前月份的付款狀態 const loadStatus = () => { const { year, month } = AppState.getYearMonth(); document.getElementById('fixed-month').textContent = `${year} 年 ${month} 月`; if (unsubStatus) unsubStatus(); unsubStatus = listenFixedStatus(year, month, (val) => { statuses = val || {}; render(); }); }; loadStatus(); AppState.on('ym-change', loadStatus); // 3. 新增模板 document.getElementById('add-fixed-btn').addEventListener('click', () => { editingId = null; modalTitle.textContent = '新增固定支出'; form.reset(); modal.classList.add('active'); setTimeout(() => nameInput.focus(), 50); }); document.getElementById('fixed-cancel-btn').addEventListener('click', () => { modal.classList.remove('active'); }); form.addEventListener('submit', async (e) => { e.preventDefault(); const payload = { name: nameInput.value.trim(), amount: Number(amountInput.value) || 0, cycle: cycleSelect.value, note: noteInput.value.trim(), type: 'family', }; if (!payload.name) return; if (editingId) { await updateFixedTemplate(editingId, payload); } else { await addFixedTemplate(payload); } modal.classList.remove('active'); }); // 4. 表格事件委派（編輯 / 刪除 / 勾選付款） tbody.addEventListener('click', async (e) => { const btn = e.target.closest('button[data-action]'); if (!btn) return; const id = btn.dataset.id; const action = btn.dataset.action; const t = templates.find((x) => x.id === id); if (!t) return; if (action === 'edit') { editingId = id; modalTitle.textContent = '編輯固定支出'; nameInput.value = t.name || ''; amountInput.value = t.amount || ''; cycleSelect.value = t.cycle || '每月'; noteInput.value = t.note || ''; modal.classList.add('active'); setTimeout(() => nameInput.focus(), 50); } else if (action === 'delete') { if (confirm(`確定要刪除「${t.name}」嗎？\n（過去月份的付款紀錄不會被刪除）`)) { await removeFixedTemplate(id); } } }); tbody.addEventListener('change', async (e) => { const checkbox = e.target; if (checkbox.dataset.action !== 'toggle-paid') return; const { year, month } = AppState.getYearMonth(); const tmplId = checkbox.dataset.id; const isPaid = checkbox.checked; await saveFixedStatus(year, month, tmplId, { status: isPaid ? '已付款' : '未付款', paidDate: isPaid ? new Date().toISOString().slice(0, 10) : '', }); }); function render() { const total = templates.reduce((s, t) => s + (Number(t.amount) || 0), 0); const pending = templates.filter((t) => { const st = statuses[t.id]; return !st || st.status !== '已付款'; }).length; document.getElementById('fixed-total').textContent = formatHKD(total); document.getElementById('fixed-pending-count').textContent = `${pending} 筆`; if (!templates.length) { tbody.innerHTML = `尚無固定支出，請點擊「新增固定支出」開始。新增後會自動套用到每個月。`; return; } tbody.innerHTML = templates.map((t) => { const st = statuses[t.id] || {}; const isPaid = st.status === '已付款'; return ` ${escapeHtml(t.name)} ${t.note ? `
${escapeHtml(t.note)}
` : ''} ${formatHKD(t.amount)} ${escapeHtml(t.cycle || '每月')} ${isPaid ? '已付款' : '未付款'} `; }).join(''); if (window.lucide && typeof window.lucide.createIcons === 'function') { window.lucide.createIcons(); } } }
